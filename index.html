function refreshSaveList(){ const select = document.getElementById("loadSelect"); const all = JSON.parse(localStorage.getItem(SAVE_KEY)) || {}; const list = Object.keys(all); select.innerHTML = ""; list.forEach(name => { const option = document.createElement("option"); option.value = name; option.textContent = name; select.appendChild(option); }); if(list.length === 0){ const option = document.createElement("option"); option.textContent = "保存なし"; select.appendChild(option); } }
